cmake_minimum_required(VERSION 3.20)
project(cix_spatial_bridge LANGUAGES CXX)
set(CIX_SPATIAL_ROOT "" CACHE PATH "Provisioned CharLS/OpenJPEG prefix")
if(NOT IS_DIRECTORY "${CIX_SPATIAL_ROOT}")
  message(FATAL_ERROR "Set CIX_SPATIAL_ROOT to the immutable codec SDK prefix")
endif()
find_path(CIX_CHARLS_INCLUDE charls/charls.h HINTS "${CIX_SPATIAL_ROOT}/usr/include" NO_DEFAULT_PATH REQUIRED)
find_path(CIX_OPENJPEG_INCLUDE openjpeg.h HINTS "${CIX_SPATIAL_ROOT}/usr/include/openjpeg-2.5" NO_DEFAULT_PATH REQUIRED)
find_library(CIX_CHARLS_LIBRARY NAMES charls HINTS "${CIX_SPATIAL_ROOT}/usr/lib/x86_64-linux-gnu" NO_DEFAULT_PATH REQUIRED)
find_library(CIX_OPENJPEG_LIBRARY NAMES openjp2 HINTS "${CIX_SPATIAL_ROOT}/usr/lib/x86_64-linux-gnu" NO_DEFAULT_PATH REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(cix_spatial_bridge SHARED cix_spatial_bridge.cpp)
target_compile_definitions(cix_spatial_bridge PRIVATE CIX_SPATIAL_BRIDGE_BUILDING)
target_include_directories(cix_spatial_bridge PRIVATE "${CIX_CHARLS_INCLUDE}" "${CIX_OPENJPEG_INCLUDE}")
target_link_libraries(cix_spatial_bridge PRIVATE "${CIX_CHARLS_LIBRARY}" "${CIX_OPENJPEG_LIBRARY}")
set_target_properties(cix_spatial_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES OUTPUT_NAME cix_spatial_bridge INSTALL_RPATH "$ORIGIN")
target_compile_options(cix_spatial_bridge PRIVATE -O3 -fno-fast-math -ffp-contract=off -fno-strict-aliasing)
include(GNUInstallDirs)
install(TARGETS cix_spatial_bridge LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES cix_spatial_bridge.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/cix")
